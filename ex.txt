nano job.sh
#!/bin/bash
echo "Job executed at $(date)" >> job.log

chmod +x job.sh
./job.sh
cat job.log

nano auto-commit.sh
#!/bin/bash
cd "$HOME/git-experiment" || exit 1
./job.sh
git add job.sh job.log
git diff --cached --quiet || git commit -m "Automatic job update"

chmod +x auto-commit.sh
./auto-commit.sh
git log --oneline

crontab -e

*/5 * * * * /bin/bash /home/christlin/git-experiment/auto-commit.sh

crontab -l
